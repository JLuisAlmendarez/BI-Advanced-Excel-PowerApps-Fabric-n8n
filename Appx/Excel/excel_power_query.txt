CONSULTA: Orders Analysis
Una fila por orden (99,441 filas). main_category = categoría del
producto más caro de la orden; permite que el panel de categorías salga
de esta misma tabla y lo filtren los mismos controles que los demás.
---------------------------------------------------------------------
let
  Items = Table.Group(#"Order Items", {"order_id"}, {
      {"items_value", each List.Sum([price]), type number},
      {"freight_value", each List.Sum([freight_value]), type number},
      {"item_count", each Table.RowCount(_), Int64.Type}}),
  Resenas = Table.Group(#"Order Reviews", {"order_id"}, {
      {"review_score", each List.Average([review_score]), type number}}),
  ItemsConCategoria = Table.ExpandTableColumn(Table.NestedJoin(#"Order Items", {"product_id"}, Products, {"product_id"}, "p", JoinKind.LeftOuter), "p", {"category"}),
  CategoriaPrincipal = Table.Group(ItemsConCategoria, {"order_id"}, {
      {"main_category", each Table.Max(_, "price")[category], type text}}),
  Base = Table.SelectColumns(Orders, {"order_id", "order_customer_id", "order_status", "purchase_date", "purchase_year", "purchase_month", "delivery_days", "delay_days", "is_late_delivery", "date_check"}),
  #"Con items" = Table.ExpandTableColumn(Table.NestedJoin(Base, {"order_id"}, Items, {"order_id"}, "x", JoinKind.LeftOuter), "x", {"items_value", "freight_value", "item_count"}),
  #"Con resenas" = Table.ExpandTableColumn(Table.NestedJoin(#"Con items", {"order_id"}, Resenas, {"order_id"}, "x", JoinKind.LeftOuter), "x", {"review_score"}),
  #"Con pagos" = Table.ExpandTableColumn(Table.NestedJoin(#"Con resenas", {"order_id"}, #"Order Payments", {"order_id"}, "x", JoinKind.LeftOuter), "x", {"total_paid", "main_payment_type"}),
  #"Con clientes" = Table.ExpandTableColumn(Table.NestedJoin(#"Con pagos", {"order_customer_id"}, Customers, {"order_customer_id"}, "x", JoinKind.LeftOuter), "x", {"customer_unique_id", "customer_city", "customer_state"}),
  #"Rango de retraso" = Table.AddColumn(#"Con clientes", "delay_bucket", each
      if [delay_days] = null then null
      else if [delay_days] <= 0 then "1. On time or early"
      else if [delay_days] <= 5 then "2. 1-5 days late"
      else if [delay_days] <= 15 then "3. 6-15 days late"
      else "4. 16+ days late", type text),
  #"Anio mes" = Table.AddColumn(#"Rango de retraso", "purchase_year_month", each Date.ToText([purchase_date], "yyyy-MM"), type text),
  #"Con categoria" = Table.ExpandTableColumn(Table.NestedJoin(#"Anio mes", {"order_id"}, CategoriaPrincipal, {"order_id"}, "x", JoinKind.LeftOuter), "x", {"main_category"})
