/* ---------------------------------------------------------------------
   1. Tablas
   --------------------------------------------------------------------- */

-- Vendedores que recibirán correos (Olist no trae correos: usa alias tuyos)
CREATE TABLE dbo.vendedor_contacto (
    seller_id  VARCHAR(32)   NOT NULL PRIMARY KEY,
    nombre     NVARCHAR(100) NOT NULL,
    correo     NVARCHAR(200) NOT NULL
);

-- Una fila por reseña de 1-2 estrellas y vendedor
CREATE TABLE dbo.seguimiento_resenas (
    id                 INT IDENTITY(1,1) PRIMARY KEY,
    review_id          VARCHAR(32)    NOT NULL,
    order_id           VARCHAR(32)    NOT NULL,
    seller_id          VARCHAR(32)    NOT NULL,
    categoria_producto NVARCHAR(100)  NULL,
    calificacion       TINYINT        NOT NULL,
    titulo             NVARCHAR(200)  NULL,
    comentario         NVARCHAR(MAX)  NULL,
    fecha_resena       DATETIME2      NOT NULL,
    correo_cliente     NVARCHAR(200)  NULL,      -- simulado, para el chat
    estado             VARCHAR(20)    NOT NULL DEFAULT 'sin_respuesta',
        -- sin_respuesta / en_contacto / cerrada
    tipo_respuesta     VARCHAR(20)    NULL,       -- comentario / accion
    respuesta          NVARCHAR(1000) NULL,
    fecha_notificacion DATETIME2      NULL,       -- la llena Power Automate
    fecha_respuesta    DATETIME2      NULL,
    fecha_cierre       DATETIME2      NULL,
    CONSTRAINT uq_resena UNIQUE (review_id, order_id, seller_id),
    CONSTRAINT ck_estado CHECK (estado IN ('sin_respuesta','en_contacto','cerrada')),
    CONSTRAINT ck_tipo   CHECK (tipo_respuesta IS NULL OR tipo_respuesta IN ('comentario','accion'))
);

-- Mensajes del chat vendedor <-> cliente
CREATE TABLE dbo.interacciones (
    id             INT IDENTITY(1,1) PRIMARY KEY,
    seguimiento_id INT            NOT NULL
        REFERENCES dbo.seguimiento_resenas(id),
    remitente      VARCHAR(10)    NOT NULL,       -- vendedor / cliente
    mensaje        NVARCHAR(MAX)  NOT NULL,
    fecha          DATETIME2      NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT ck_remitente CHECK (remitente IN ('vendedor','cliente'))
);

CREATE INDEX ix_seg_seller_estado ON dbo.seguimiento_resenas (seller_id, estado);
CREATE INDEX ix_int_seguimiento   ON dbo.interacciones (seguimiento_id, fecha);


/* ---------------------------------------------------------------------
   2. Carga desde gold
   Antes: con una actividad Copy del pipeline, copia del Lakehouse a esta
   base tres tablas de staging (copia completa, sin filtro):
     order_reviews -> dbo.stg_order_reviews
     order_items   -> dbo.stg_order_items
     products      -> dbo.stg_products
   --------------------------------------------------------------------- */

INSERT INTO dbo.seguimiento_resenas
    (review_id, order_id, seller_id, categoria_producto,
     calificacion, titulo, comentario, fecha_resena)
SELECT
    r.review_id,
    r.order_id,
    v.seller_id,
    v.categoria_producto,
    r.review_score,
    r.review_comment_title,
    r.review_comment_message,
    r.review_creation_date
FROM dbo.stg_order_reviews AS r
JOIN (
    -- un pedido puede tener varias líneas del mismo vendedor
    SELECT i.order_id, i.seller_id, MAX(p.category) AS categoria_producto
    FROM dbo.stg_order_items AS i
    LEFT JOIN dbo.stg_products AS p
        ON p.product_id = i.product_id
    GROUP BY i.order_id, i.seller_id
) AS v
    ON v.order_id = r.order_id
WHERE r.review_score <= 2;

-- Revisión rápida
SELECT COUNT(*) AS filas, COUNT(DISTINCT seller_id) AS vendedores
FROM dbo.seguimiento_resenas;
